🔧 Supabase RLS Fix

RLS (Row Level Security) blokerer coachen fra at læse klienters data og gemme planer. Kør hele denne SQL i Supabase → SQL Editor:

-- ═══ Coach RLS Fix – kør alt på én gang ═══

-- Sikrer at client_data kun har én række per bruger (kræves af upsert)
ALTER TABLE client_data DROP CONSTRAINT IF EXISTS client_data_user_id_key;
ALTER TABLE client_data ADD CONSTRAINT client_data_user_id_key UNIQUE (user_id);

-- profiles: coach ser alle klienter
DROP POLICY IF EXISTS "profiles_access" ON profiles;
CREATE POLICY "profiles_access" ON profiles FOR ALL
  USING (auth.uid() = id OR (SELECT email FROM auth.users WHERE id = auth.uid()) = 'hamzaelouaz@gmail.com')
  WITH CHECK (auth.uid() = id OR (SELECT email FROM auth.users WHERE id = auth.uid()) = 'hamzaelouaz@gmail.com');

-- intake_forms: coach kan læse alle opstartsskemaer
DROP POLICY IF EXISTS "intake_forms_access" ON intake_forms;

  WITH CHECK (auth.uid() = user_id OR (SELECT email FROM auth.users WHERE id = auth.uid()) = 'hamzaelouaz@gmail.com');
